% Règles (IDB)
chemin(X, Y) :- troncon(_, X, Y, _).
chemin(X, Z) :- troncon(_, X, Y, _), chemin(Y, Z).

% Règle pour les trajets de randonnées avec un nombre impair de tronçons
trajet_circuit(X, Z) :- 
    chemin(X, Z),
    chemin(X, Y), 
    chemin(Y, Z),
    X = Z.
